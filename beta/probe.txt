        async _addToCart() {
            if (!this.currentProduct?.id) return;

            const wcAjaxUrl = this.config.wc_ajax_url;
            if (!wcAjaxUrl) {
                console.warn('[FWAP] WooCommerce AJAX URL not configured');
                return;
            }

            this.el.cartBtn?.classList.add('fwap-loading');

            try {
                const formData = new FormData();
                formData.append('product_id', this.currentProduct.id);
                formData.append('quantity', 1);

                const response = await fetch(wcAjaxUrl.replace('%%endpoint%%', 'add_to_cart'), {
                    method: 'POST',
                    body: formData,
                    credentials: 'same-origin'
                });

                const data = await response.json();

                if (data.error) {
                    // B2B sites may require login - redirect to product page
                    if (data.product_url) {
                        window.location.href = data.product_url;
                        return;
                    }
                    throw new Error(typeof data.error === 'string' ? data.error : 'Unable to add to cart');
                }

                this.el.cartBtn?.classList.add('fwap-added');
                this._showStatus(this.config.i18n?.added || 'Added to cart', 'success');

                // Trigger WooCommerce cart update event
                document.body.dispatchEvent(new CustomEvent('added_to_cart', {
                    detail: { product_id: this.currentProduct.id }
                }));

                // Reset button after delay
                setTimeout(() => {
                    this.el.cartBtn?.classList.remove('fwap-added');
                }, 2000);

            } catch (error) {
                console.error('[FWAP] Add to cart failed:', error);
                this._showStatus(this.config.i18n?.error || 'Error', 'error');
            } finally {
                this.el.cartBtn?.classList.remove('fwap-loading');
            }
        }

                if (json && json.code) {
                    err.code = json.code;
                }
                throw err;
            }

            // Check for success response (standard format)
            if (json && (json.success === false || json.code)) {
                const err = new Error(json.message || 'API Error');
                if (json.code) {
                    err.code = json.code;
                }
                throw err;
            }

            // Normalize response format
            return this._normalizeProductData(json, productId);
